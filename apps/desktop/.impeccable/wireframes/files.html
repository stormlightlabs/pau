204 /// Spawns the agent inside the workspace sandbox and waits for its
205 /// extension to report ready. A session that misses the wait is killed
206 /// and reported as failed, so the bay never shows a session that cannot
207 /// take a prompt.
208 pub async fn start(&self, request: StartRequest) -> Result<Session, StartError> {
209 let agent = self.agents.get(&request.agent).ok_or(StartError::UnknownAgent)?;
210 let command = agent.launch_command(&request.workspace)?;
211
212 tracing::info!(agent = %request.agent, workspace = %request.workspace, "starting session");
213 let child = self.sandbox.spawn(command)?;
214 let ready = timeout(request.ready_timeout.unwrap_or(READY_TIMEOUT), child.wait_ready());
215 match ready.await {
216 Ok(Ok(())) => Ok(Session::running(child)),
217 Ok(Err(err)) => Err(StartError::Extension(err)),
218 Err(_) => {
219 child.kill().await?;
220 self.events.send(SessionEvent::failed(&request, "extension never reported ready")).ok();
221 Err(StartError::NotReady(READY_TIMEOUT))
222 }
223 }
224 }
